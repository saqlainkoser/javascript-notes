function fetchData(){

    return new Promise((resolve,reject)=>{

        setTimeout(()=>{

            resolve("User data")

            console.log("Data Received")

        },3000)

    })

}

function processdata(data){

    return new Promise((resolve,reject)=>{

           setTimeout(()=>{

            resolve(`${data} processed successfully`)

            console.log("Data Processed")

        },2000)

    })

}
