//PROCESS DATA ,FETCH DATA EXAMPLE  
  
  
 

function fetchData() {

  return new Promise((resolve) => {

    setTimeout(() => {

      console.log("📥 Data fetched");

      resolve("User Data");

    }, 2000);

  });

}

function processData(data) {

  return new Promise((resolve) => {

    setTimeout(() => {

      console.log("⚙️ Data processed");

      resolve(`${data} successfully processed`);

    }, 1500);

  });

}

async function main() {

  console.log("🚀 Start");

  const data = await fetchData(); // wait for fetchData

  const result = await processData(data); // wait for processData

  console.log("✅ Result:", result);

  console.log("🏁 End");

}

main();
